package de.regelsuche.evolution;

import de.regelsuche.ast.BinaryExpr;
import de.regelsuche.ast.BinaryOperator;

import de.regelsuche.retention.RetainedOperation;

/** Declared finite procedures, not a learned strategy or a second search algorithm. */
final class StructuralConstructionPrograms {
    enum Operation { EXPOSE_SQUARE, COMPLETE_SQUARE, FACTOR_DIFFERENCE, FACTOR_PARTITION, RESIDUAL_BALANCE,
        BINARY_QUARTIC, RESIDUAL_EXPANDED, AUXILIARY_COMPLEMENT, AUXILIARY_FACTOR, RECOGNIZE_SQUARE,
        RATIONAL_EXPOSE_SQUARE, RECOGNIZE_RATIONAL_SQUARE, RECOGNIZE_METERED_SQUARE,
        POLYNOMIAL_CANONICAL, POLYNOMIAL_EXPANDED, POLYNOMIAL_COMMON_FACTOR, POLYNOMIAL_UNIVARIATE_FACTOR,
        POLYNOMIAL_HORNER, POLYNOMIAL_DIRECT }
    record Report(List<String> candidates, long work) { Report { candidates = List.copyOf(candidates); } }
    private static final long LIMIT = 100_000;
    private StructuralConstructionPrograms() { }

    static Report generate(Operation operation, Expr source) {
        var work = operation == Operation.POLYNOMIAL_HORNER ? new Work(4_096) : new Work();
        try {
            if (operation == Operation.POLYNOMIAL_HORNER) {
                var candidate = SparsePolynomialHorner.propose(source, work::add);
                return new Report(candidate.stream()
                    .map(expression -> ExpressionFormatter.formatMeasured(expression, work::add)).toList(), work.units());
            }
            var domain = new CheckedSchemaSupport.Work();
            try {
                CheckedSchemaSupport.domain(source, CheckedLearnedSchemaModel.Bounds.defaults(), domain);
            } finally {
                work.add(domain.units);
            }
            String text = ExpressionFormatter.format(source); work.add(text.length());
            var candidates = switch (operation) {
                case EXPOSE_SQUARE -> apply(new ExactMonomialSquareExposureOperator(8), text, work);
                case COMPLETE_SQUARE -> apply(new SumOfSquaresCompletionOperator(), text, work);
                case FACTOR_DIFFERENCE -> factorDifference(text, work);
                case FACTOR_PARTITION -> partitionFactors(source, text, work);
                case RESIDUAL_BALANCE -> residualBalance(source, text, work, false);
                case BINARY_QUARTIC -> QuarticConstruction.generate(text, LIMIT - work.units, work::add);
                case RESIDUAL_EXPANDED -> residualBalance(source, text, work, true);
                case AUXILIARY_COMPLEMENT -> explorePolynomial(AlgebraicConstruction.Operation.ADDITIVE_COMPLEMENT, text, work);
                case AUXILIARY_FACTOR -> explorePolynomial(AlgebraicConstruction.Operation.EXACT_FACTOR_SPLIT, text, work);
                case RECOGNIZE_SQUARE -> recognizeSquare(text, work, false, false);
                case RATIONAL_EXPOSE_SQUARE -> apply(new ExactMonomialSquareExposureOperator(8, work::add), text, work);
                case RECOGNIZE_RATIONAL_SQUARE -> recognizeSquare(text, work, true, true);
                case RECOGNIZE_METERED_SQUARE -> recognizeSquare(text, work, false, true);
                case POLYNOMIAL_CANONICAL, POLYNOMIAL_EXPANDED, POLYNOMIAL_COMMON_FACTOR, POLYNOMIAL_UNIVARIATE_FACTOR ->
                    PolynomialSimplificationPrograms.generate(operation, source, LIMIT - work.units, work::add);
                case POLYNOMIAL_HORNER -> throw new IllegalStateException("Horner handled before legacy generation");
                case POLYNOMIAL_DIRECT -> throw new IllegalStateException("direct portfolio requires a request budget");
            };
            return new Report(candidates, work.units);
        } catch (Limit | IllegalArgumentException unsupported) {
            return new Report(List.of(), work.units);
        }
    }

    private static List<String> apply(HypothesisOperator operator, String source, Work work) {
        work.add(source.length() + 1L);
        var candidates = operator.generateCandidates(source);
        var result = new ArrayList<String>();
        for (var candidate : candidates) {
            work.add(candidate.transformedExpression().length() + 1L);
            if (candidate.assumptions().isEmpty()) result.add(candidate.transformedExpression());
        }
        return List.copyOf(result);
    }
    private static final class DifferenceOfSquares {
        static final List<RewriteRule> RULES = differenceOfSquaresRules();
    }

    private static List<RewriteRule> differenceOfSquaresRules() {
        return AstRewriteTransformationEngine.allBuiltInRules().stream()
            .filter(rule -> rule.id().equals("ast_square_difference_factor")).toList();
    }

    private static List<String> factorDifference(String source, Work work) {
        var rules = RetainedOperation.isObserved() ? differenceOfSquaresRules() : DifferenceOfSquares.RULES;
        return apply(new AstRewriteTransformationEngine(rules)::transform, source, work);
    }

    private static List<String> explorePolynomial(AlgebraicConstruction.Operation operation, String source, Work work) {
        long remaining = LIMIT - work.units;
        if (remaining <= 0) throw new Limit();
        var exploration = new AlgebraicConstruction().explorePolynomial(operation, source, List.of(), 4, remaining);
        work.add(exploration.work());
        return exploration.results().stream().filter(AlgebraicConstruction.Result::generated)
            .flatMap(result -> result.expression().stream()).toList();
    }

    private static List<String> recognizeSquare(String source, Work work, boolean rational, boolean metered) {
        var delegate = metered ? new ConservativeCompleteSquareHypothesisOperator(6, true, rational, work::add)
            : new ConservativeCompleteSquareHypothesisOperator(6, true);
        HypothesisOperator measured = subtree -> {
            work.add(subtree.length() + 1L);
            var candidates = delegate.generateCandidates(subtree);
            for (var candidate : candidates) work.add(candidate.transformedExpression().length() + 1L);
            return candidates;
        };
        return apply(new SubtreeHypothesisOperator(measured, 8), source, work);
    }

    /** Construct exact monomial factors instead of introducing a potentially zero denominator. */
    private static List<String> partitionFactors(Expr expression, String source, Work work) {
        if (!(expression instanceof BinaryExpr power) || power.operator() != BinaryOperator.POW
                || !(power.right() instanceof NumberExpr exponent)) return List.of();
        var value = exponent.value();
        if (!value.denominator().equals(java.math.BigInteger.ONE) || value.numerator().signum() < 1
                || value.numerator().compareTo(java.math.BigInteger.valueOf(12)) > 0) return List.of();
        int n = value.numerator().intValueExact();
        if (n < 2) return List.of();
        String auxiliary = "(" + ExpressionFormatter.format(power.left()) + ")^" + (n / 2);
        var result = new AlgebraicConstruction().construct(new AlgebraicConstruction.Request(
            AlgebraicConstruction.Operation.EXACT_FACTOR_SPLIT, source, auxiliary, List.of(), LIMIT - work.units));
        work.add(result.work());
        return result.expression().stream().toList();
    }

    private static List<String> residualBalance(Expr expression, String source, Work work, boolean expanded) {
        if (!(expression instanceof BinaryExpr root)
                || (expanded ? root.operator() != BinaryOperator.ADD : root.operator() != BinaryOperator.MUL)) return List.of();
        var prepared = prepareResidual(source, work, expanded);
        if (prepared == null) return List.of();
        var results = prepared.composer().compose(prepared.expression(), prepared.parts(), prepared.effects(), prepared.parts().size() / 2, 8);
        work.add(results.size());
        return results.stream().map(ExactPolynomialResidualComposer.Composition::candidateExpression).toList();
    }

    record PreparedResidual(String expression, List<ExactPolynomialResidualComposer.SourceComponent> parts,
                            List<ExactPolynomialResidualComposer.Effect> effects, ExactPolynomialResidualComposer composer) { }

    static PreparedResidual prepareResidual(String source, Work work, boolean productWork) {
        var preparation = new ResidualPreparation(source, work, productWork);
        while (!preparation.done) preparation.advance();
        return preparation.result;
    }

    /** Source-owned checked prefix. Each pull prepares the components or derives one local effect. */
    static final class ResidualPreparation {
        final String source;
        final Work work;
        final boolean productWork;
        boolean done;
        PreparedResidual result;
        private String expression;
        private ExactPolynomialResidualComposer composer;
        private List<ExactPolynomialResidualComposer.SourceComponent> parts;
        private final List<ExactPolynomialResidualComposer.Effect> effects = new ArrayList<>();
        private int left, right = 1;
        private boolean reflected;

        ResidualPreparation(String source, Work work, boolean productWork) {
            this.source = source;
            this.work = work;
            this.productWork = productWork;
        }

        void advance() {
            if (done) throw new IllegalStateException("residual preparation is complete");
            if (parts == null) {
                var exact = productWork ? ExactPolynomialAnalysis.withProductWork(work::add) : new ExactPolynomialAnalysis(work::add);
                var polynomial = exact.normalForm(source);
                int count = polynomial.termCount();
                if (count < 4 || count > 6 || count % 2 != 0) { done = true; return; }
                expression = exposeAll(polynomial.toCanonicalString(), work, productWork);
                composer = productWork ? ExactPolynomialResidualComposer.withProductWork(work::add) : new ExactPolynomialResidualComposer(work::add);
                parts = composer.additiveComponents(expression);
                done = parts.size() != count || !allSquares(parts, work);
                return;
            }
            effects.add(effect(List.of(parts.get(left), parts.get(right)), reflected, left + "-" + right, composer, work));
            reflected = !reflected;
            if (!reflected && ++right == parts.size()) {
                right = ++left + 1;
                if (right == parts.size()) {
                    result = new PreparedResidual(expression, parts, List.copyOf(effects), composer);
                    done = true;
                }
            }
        }
    }

    private static String exposeAll(String source, Work work, boolean includeProductWork) {
        String current = source;
        var exposure = new ExactMonomialSquareExposureOperator(1);
        var exact = includeProductWork ? ExactPolynomialAnalysis.withProductWork(work::add) : new ExactPolynomialAnalysis(work::add);
        for (int step = 0; step < 12; step++) {
            var candidates = apply(exposure, current, work);
            if (candidates.isEmpty()) return current;
            String next = candidates.getFirst(); exact.requireEquivalent(current, next); current = next;
        }
        throw new IllegalArgumentException("square preparation did not converge within its bound");
    }
    private static boolean allSquares(List<ExactPolynomialResidualComposer.SourceComponent> parts, Work work) {
        var parser = new ExpressionParser();
        for (var part : parts) {
            work.add(part.expression().length());
            if (!StructuralTaskObjective.isSquare(parser.parseExactTerm(part.expression()).expression())) return false;
        }
        return true;
    }
    private static ExactPolynomialResidualComposer.Effect effect(
            List<ExactPolynomialResidualComposer.SourceComponent> pair, boolean reflect, String id,
            ExactPolynomialResidualComposer composer, Work work) {
        String right = pair.get(1).expression();
        var rules = new ArrayList<String>(); var applications = new ArrayList<String>();
        if (reflect) {
            var symmetry = only(new SquareBaseSignSymmetryOperator().generateCandidates(right));
            work.add(right.length() + symmetry.transformedExpression().length());
            right = symmetry.transformedExpression(); rules.addAll(symmetry.primitiveRuleIds()); applications.add(symmetry.applicationKey());
        }
        String input = "(" + pair.get(0).expression() + ")+(" + right + ")";
        var completion = only(new SumOfSquaresCompletionOperator().generateCandidates(input));
        work.add(input.length() + completion.transformedExpression().length());
        rules.addAll(completion.primitiveRuleIds()); applications.add(completion.applicationKey());
        return composer.effect((reflect ? "reflected-" : "plus-") + id, pair, completion.transformedExpression(),
            completedSquare(completion.transformedExpression(), work), List.of(), rules, applications);
    }
    private static String completedSquare(String source, Work work) {
        var pending = new ArrayDeque<Expr>(); pending.push(new ExpressionParser().parseExactTerm(source).expression());
        while (!pending.isEmpty()) {
            Expr node = pending.pop(); work.add(1);
            if (!(node instanceof BinaryExpr binary)) continue;
            if (StructuralTaskObjective.isSquare(binary) && binary.left() instanceof BinaryExpr base
                    && (base.operator() == BinaryOperator.ADD || base.operator() == BinaryOperator.SUB))
                return ExpressionFormatter.format(binary);
            pending.push(binary.right()); pending.push(binary.left());
        }
        throw new IllegalArgumentException("completion did not retain a structured square");
    }
    private static Transformation only(List<Transformation> results) {
        if (results.size() != 1) throw new IllegalArgumentException("one bound completion required");
        return results.getFirst();
    }
    static final class Work {
        private final long limit;
        private final boolean strict;
        private long units;
        Work() { limit = LIMIT; strict = false; }
        Work(long limit) {
            this.limit = requireLimit(limit);
            this.strict = true;
        }
        static long requireLimit(long limit) {
            if (limit < 0 || limit > LIMIT) throw new IllegalArgumentException("work limit outside the finite grammar");
            return limit;
        }
        long units() { return units; }
        void add(long count) {
            long next = Math.addExact(units, count);
            if (strict && next > limit) throw new Limit();
            units = next;
            if (units > limit) throw new Limit();
        }
    }
    static final class Limit extends RuntimeException { }
}
